cmake_minimum_required(VERSION 3.21)
project(controlled_triad LANGUAGES CXX HIP)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_HIP_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(NOT CMAKE_HIP_ARCHITECTURES)
  set(CMAKE_HIP_ARCHITECTURES gfx942)
endif()
find_path(ROCTX_INCLUDE_DIR rocprofiler-sdk-roctx/roctx.h HINTS "$ENV{ROCM_PATH}/include" /opt/rocm/include REQUIRED)
find_library(ROCTX_LIBRARY rocprofiler-sdk-roctx HINTS "$ENV{ROCM_PATH}/lib" /opt/rocm/lib REQUIRED)
add_executable(triad_worker triad_worker.hip)
target_include_directories(triad_worker PRIVATE ${ROCTX_INCLUDE_DIR})
target_link_libraries(triad_worker PRIVATE ${ROCTX_LIBRARY})
target_compile_options(triad_worker PRIVATE -Wall -Wextra)
